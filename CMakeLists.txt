cmake_minimum_required(VERSION 3.16)
project(ama10 C)
option(AMA10_ENABLE_OPENCL "Build the OpenCL GPU inference backend" OFF)
option(AMA10_ENABLE_PARQUET "Build the C11 native Parquet shard processor (requires scripts/fetch_duckdb.ps1)" OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(ama10_core src/model.c src/encoder.c src/image.c src/image_gen.c src/audio.c src/tokenizer.c src/checkpoint.c src/multimodal_checkpoint.c src/dataset.c src/eval.c src/lora.c src/tools.c src/q8.c)
target_include_directories(ama10_core PUBLIC include)
if(NOT MSVC)
  target_link_libraries(ama10_core PUBLIC m)
endif()
add_executable(ama10 src/main.c)
target_link_libraries(ama10 PRIVATE ama10_core)
add_executable(ama10_image_caption_train src/image_caption_train.c)
target_link_libraries(ama10_image_caption_train PRIVATE ama10_core)

target_link_libraries(ama10_image_caption_generate PRIVATE ama10_core)

target_link_libraries(ama10_image_gen_generate PRIVATE ama10_core)

target_link_libraries(ama10_supervisor PRIVATE ama10_core)
if(WIN32)
  target_link_libraries(ama10_supervisor PRIVATE winhttp)
endif()
if(WIN32 AND AMA10_ENABLE_PARQUET)
  set(AMA10_DUCKDB_DIR "${CMAKE_SOURCE_DIR}/build/deps/duckdb-2.0.0" CACHE PATH "Pinned DuckDB C API headers")
  if(NOT EXISTS "${AMA10_DUCKDB_DIR}/duckdb.h")
    message(FATAL_ERROR "DuckDB C API header is missing; run scripts/fetch_duckdb.ps1 first")
  endif()
  add_executable(ama10_parquet_ingest src/pipeline_ledger.c src/parquet_ingest.c src/parquet_ingest_cli.c)
  target_include_directories(ama10_parquet_ingest PRIVATE include "${AMA10_DUCKDB_DIR}")
  target_link_libraries(ama10_parquet_ingest PRIVATE bcrypt)
endif()
add_executable(ama10_tool_runtime src/tool_runtime.c)

  target_link_libraries(ama10_swanlab PRIVATE winhttp)

  add_executable(ama10_text_split src/text_split.c)
  target_compile_definitions(ama10_text_split PRIVATE AMA10_TEXT_SPLIT_CLI=1)
endif()

  add_executable(ama10_fetch_shard src/pipeline_ledger.c src/shard_fetch.c src/shard_fetch_cli.c)
  target_link_libraries(ama10_fetch_shard PRIVATE winhttp bcrypt)
endif()
add_executable(ama10_score_text src/text_score.c)
target_link_libraries(ama10_score_text PRIVATE ama10_core)
if(AMA10_ENABLE_OPENCL)
  find_package(OpenCL REQUIRED)
  target_sources(ama10_core PRIVATE src/opencl_backend.c)
  target_compile_definitions(ama10_core PUBLIC AMA10_ENABLE_OPENCL=1)
  target_link_libraries(ama10_core PUBLIC OpenCL::OpenCL)

  target_compile_definitions(ama10_opencl_bench PRIVATE AMA10_ENABLE_OPENCL=1)
  target_link_libraries(ama10_opencl_bench PRIVATE ama10_core OpenCL::OpenCL)
endif()
